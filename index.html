将鼠标悬浮在代码块右上角，即可看到 Copy 按钮：

import math

def poisson_pmf(k: int, lambd: float) -> float:
    """计算泊松分布单点概率 P(X = k)"""
    if k < 0 or lambd <= 0:
        raise ValueError("k 必须是非负整数，lambda 必须大于 0")
    return (lambd ** k * math.exp(-lambd)) / math.factorial(k)

# 测试计算: 平均每分钟到达 4 人，实际到达 2 人的概率
print(f"P(X = 2) = {poisson_pmf(2, 4.0):.4f}")
